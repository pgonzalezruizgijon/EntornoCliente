1. Estructura básica de Jasmine

Jasmine sirve para comprobar automáticamente si una función, objeto o clase funciona como esperamos.

La estructura mental es: describe agrupa tests, it describe una prueba, expect comprueba el resultado. 
describe("Nombre del bloque de pruebas", function() {

  it("debería hacer algo concreto", function() {
    let resultado = sumar(2, 3);
    expect(resultado).toEqual(5);
  });

});

Ejemplo con una función

function sumar(a, b) {
  return a + b;
}

describe("Función sumar", function() {

  it("debería sumar dos números", function() {
    expect(sumar(2, 3)).toEqual(5);
  });

});
En el examen, normalmente te dan funciones o clases y tú escribes los tests con expect(...).matcher(...).

2. Matchers principales traducidos

Un matcher es la comprobación que va después de expect().

Matcher Sirve para... Ejemplo
not Niega la comprobación. expect(x).not.toEqual(5);
toEqual() Comprueba igualdad de valores, arrays y objetos. expect([1,2]).toEqual([1,2]);
toBeDefined() Comprueba que algo está definido. expect(resultado).toBeDefined();
toBeUndefined() Comprueba que algo es undefined. expect(valor).toBeUndefined();
toBeTrue() Comprueba que es true. expect(esMayorEdad(20)).toBeTrue();
toBeFalse() Comprueba que es false. expect(esMayorEdad(15)).toBeFalse();
toBeNull() Comprueba que es null. expect(usuario).toBeNull();
toBeNaN() Comprueba que es Not a Number. expect(Number("hola")).toBeNaN();
toContain() Comprueba que un array o string contiene algo. expect(["PHP","JS"]).toContain("JS");
toHaveSize() Comprueba tamaño de array/string/objeto. expect([1,2]).toHaveSize(2);
toBeGreaterThan() Mayor que. expect(10).toBeGreaterThan(5);
toBeGreaterThanOrEqual() Mayor o igual que. expect(18).toBeGreaterThanOrEqual(18);
toBeLessThan() Menor que. expect(3).toBeLessThan(10);
toBeLessThanOrEqual() Menor o igual que. expect(120).toBeLessThanOrEqual(120);
toBeCloseTo() Compara decimales con precisión. expect(3.1416).toBeCloseTo(3.142, 3);
toBeInstanceOf() Comprueba si algo es instancia de una clase. expect(new Alumno()).toBeInstanceOf(Alumno);
toThrowError() Comprueba que una función lanza error. expect(function(){ obj.metodo(); }).toThrowError();

3. Ejemplos rápidos por tipo de dato

Números

expect(edad).toBeGreaterThanOrEqual(18);
expect(altura).toBeLessThanOrEqual(210);
expect(media).toBeCloseTo(7.33, 2);

Booleanos

expect(esMayorEdad(20)).toBeTrue();
expect(esMayorEdad(15)).toBeFalse();

Arrays

let lenguajes = ["PHP", "JS", "CSS"];
expect(lenguajes).toContain("JS");
expect(lenguajes).toHaveSize(3);
expect(lenguajes).toEqual(["PHP", "JS", "CSS"]);

Objetos

let alumno = { nombre: "Ana", edad: 20 };
expect(alumno).toEqual({ nombre: "Ana", edad: 20 });
expect(alumno.nombre).toEqual("Ana");
expect(alumno.edad).toBeGreaterThanOrEqual(18);

4. Tests con objetos y clases

Clase de ejemplo

class Alumno {
  constructor(nombre, edad) {
    this.nombre = nombre;
    this.edad = edad;
  }

  esMayorEdad() {
    return this.edad >= 18;
  }

  cambiarEdad(edad) {
    if (edad < 0) {
      throw new Error("La edad no puede ser negativa");
    }
    this.edad = edad;
  }
}

Tests de la clase

describe("Clase Alumno", function() {

  it("debería crear un alumno", function() {
    let alumno = new Alumno("Luis", 20);

    expect(alumno).toBeDefined();
    expect(alumno).toBeInstanceOf(Alumno);
    expect(alumno.nombre).toEqual("Luis");
    expect(alumno.edad).toEqual(20);
  });

  it("debería comprobar si es mayor de edad", function() {
    let alumno = new Alumno("Ana", 19);

    expect(alumno.esMayorEdad()).toBeTrue();
  });

  it("debería comprobar si es menor de edad", function() {
    let alumno = new Alumno("Pepe", 16);

    expect(alumno.esMayorEdad()).toBeFalse();
  });

  it("debería cambiar la edad", function() {
    let alumno = new Alumno("Marta", 20);

    alumno.cambiarEdad(25);

    expect(alumno.edad).toEqual(25);
  });

});

5. Testear errores con toThrowError

Cuando una función debe lanzar un error, Jasmine lo comprueba con toThrowError().

Importante: dentro de expect() se mete una función anónima. No se llama directamente al método. 
// BIEN
expect(function() {
  alumno.cambiarEdad(-5);
}).toThrowError("La edad no puede ser negativa");

// MAL
expect(alumno.cambiarEdad(-5)).toThrowError();

Ejemplo completo

it("debería lanzar error si la edad es negativa", function() {
  let alumno = new Alumno("Luis", 20);

  expect(function() {
    alumno.cambiarEdad(-1);
  }).toThrowError("La edad no puede ser negativa");
});

6. Ejercicios típicos resueltos

Ejercicio 1: función mayor de edad

function esMayorEdad(edad) {
  return edad >= 18;
}

describe("Función esMayorEdad", function() {

  it("debería devolver true con 18", function() {
    expect(esMayorEdad(18)).toBeTrue();
  });

  it("debería devolver false con 17", function() {
    expect(esMayorEdad(17)).toBeFalse();
  });

});

Ejercicio 2: función que devuelve un array

function obtenerPares(numeros) {
  return numeros.filter(n => n % 2 === 0);
}

describe("Función obtenerPares", function() {

  it("debería devolver solo los pares", function() {
    expect(obtenerPares([1,2,3,4])).toEqual([2,4]);
  });

  it("debería devolver un array de tamaño 2", function() {
    expect(obtenerPares([1,2,3,4])).toHaveSize(2);
  });

  it("debería contener el número 4", function() {
    expect(obtenerPares([1,2,3,4])).toContain(4);
  });

});

Ejercicio 3: objeto producto

let producto = {
  nombre: "Ratón",
  precio: 15,
  stock: 10
};

describe("Objeto producto", function() {

  it("debería estar definido", function() {
    expect(producto).toBeDefined();
  });

  it("debería tener precio mayor que 0", function() {
    expect(producto.precio).toBeGreaterThan(0);
  });

  it("debería tener stock menor o igual que 100", function() {
    expect(producto.stock).toBeLessThanOrEqual(100);
  });

});

7. Chuleta final para el examen

// Estructura básica
describe("Bloque", function() {
  it("debería hacer algo", function() {
    expect(resultado).toEqual(esperado);
  });
});

// Negar
expect(valor).not.toEqual(5);

// Booleanos
expect(valor).toBeTrue();
expect(valor).toBeFalse();

// Números
expect(valor).toBeGreaterThan(3);
expect(valor).toBeGreaterThanOrEqual(18);
expect(valor).toBeLessThan(10);
expect(valor).toBeLessThanOrEqual(120);
expect(valor).toBeCloseTo(3.14, 2);

// Undefined / null / NaN
expect(valor).toBeDefined();
expect(valor).toBeUndefined();
expect(valor).toBeNull();
expect(valor).toBeNaN();

// Arrays y strings
expect(array).toContain(elemento);
expect(string).toContain("texto");
expect(array).toHaveSize(3);
expect(array).toEqual([1,2,3]);

// Objetos y clases
expect(objeto).toEqual({ nombre: "Ana" });
expect(objeto).toBeInstanceOf(Clase);

// Errores
expect(function() {
  objeto.metodoIncorrecto();
}).toThrowError("Mensaje del error");
